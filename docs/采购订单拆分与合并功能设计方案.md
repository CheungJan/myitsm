## 一、需求概述

### 1.1 背景
当前采购订单创建流程为"需求单 → 订单 → 供应商"一对一模式，存在以下业务限制：
1. **无法拆单**：同一需求单的商品不能分配给多个供应商
2. **无法并单**：多个需求单的同类商品不能合并采购

### 1.2 需求场景

#### 场景A：拆单（高优先级）
```
需求单PP000180：电源×100个
├── 供应商A供货60个（单价低，交期长）
└── 供应商B供货40个（单价高，现货）
→ 生成2个采购订单
```

#### 场景B：并单（中优先级）
```
部门A申请：电源×50个 → 2024-01-01
部门B申请：电源×30个 → 2024-01-02
部门C申请：网线×20个 → 2024-01-03

合并采购：
├── 电源×80个 → 统一发给供应商甲
└── 网线×20个 → 发给供应商乙
```

---

## 二、方案选型

### 2.1 方案对比

| 方案 | 描述 | 优点 | 缺点 | 推荐度 |
|------|------|------|------|--------|
| **方案A** | 一个订单明细行支持多供应商 | 结构简单 | 破坏现有数据模型 | ❌ |
| **方案B** | 拆单生成多个订单 | 符合现有架构 | 订单数量增加 | ✅ 推荐 |
| **方案C** | 新增子订单概念 | 灵活度高 | 改动量大 | ⚠️ |

### 2.2 选定方案：方案B（多订单模式）

**核心设计原则**：
- 保持 `tpc12_register` 和 `tpc13_registerdt` 结构不变
- 通过 `tpc20_requisition_order_link` 建立多对多关系
- 一个需求明细可关联多个订单（拆单）
- 一个订单可关联多个需求明细（并单）

### 3.4 后端实现

#### 3.4.1 Service层新增方法

```python
# app/services/procurement_service.py

class PurchaseRegisterService:
    
    @staticmethod
    def batch_create(
        orders_data: list[dict[str, Any]],
        creator: str
    ) -> dict[str, Any]:
        """
        批量创建采购订单（支持拆单场景）
        
        Args:
            orders_data: 订单列表，每个订单包含suppliercd和details
            creator: 创建人
            
        Returns:
            创建的订单ID列表
        """
        created_orders = []
        
        # 1. 预校验：汇总所有需求行的数量
        req_line_qty_map = defaultdict(float)
        for order in orders_data:
            for d in order.get("details", []):
                key = (d.get("ref_pcplanid"), d.get("ref_pclineno"))
                req_line_qty_map[key] += d.get("rgsqty", 0)
        
        # 2. 校验总数量不超过可用余额
        for (pcplanid, pclineno), total_qty in req_line_qty_map.items():
            if pcplanid and pclineno is not None:
                available = PurchasePlanRepository.get_available_qty(
                    str(pcplanid), int(pclineno)
                )
                if total_qty > available:
                    raise ValueError(
                        f"需求 {pcplanid} 行 {pclineno} "
                        f"总采购数量({total_qty})超过可用余额({available})"
                    )
        
        # 3. 逐个创建订单
        for order_data in orders_data:
            record = PurchaseRegisterRepository.create(
                order_data, creator
            )
            rgstbillid = record.rgstbillid
            
            # 创建明细
            for i, d in enumerate(order_data.get("details", []), start=1):
                PurchaseRegisterRepository.add_detail(
                    rgstbillid=rgstbillid, lineno=i, data=d
                )
            
            # 创建TPC20关联
            link_details = [
                {**d, "rgstbillid": rgstbillid, "rgstlineno": i}
                for i, d in enumerate(order_data.get("details", []), start=1)
                if d.get("ref_pcplanid") and d.get("ref_pclineno") is not None
            ]
            if link_details:
                RequisitionOrderLinkRepository.create_links(link_details)
            
            created_orders.append(rgstbillid)
        
        db.session.commit()
        return {
            "created_orders": created_orders,
            "count": len(created_orders)
        }
```

def create_batch_orders():

    json_data = request.get_json(silent=True) or {}
    
    # 校验请求结构
    if "orders" not in json_data or not isinstance(json_data["orders"], list):
        return error_response(message="请求格式错误：缺少orders数组", code=400)
    
    orders = json_data["orders"]
    if len(orders) == 0:
        return error_response(message="orders不能为空", code=400)
    
    if len(orders) > 10:
        return error_response(message="单次最多创建10个订单", code=400)
    
    user_cd: str = g.current_user
    
    try:
        result = PurchaseRegisterService.batch_create(orders, user_cd)
        return success_response(
            data=result, 
            message=f"成功创建{result['count']}个采购订单", 
            code=201

    except ValueError as e:
        return error_response(message=str(e), code=400)
    except Exception as e:
        db.session.rollback()
        return error_response(message=f"创建失败: {str(e)}", code=500)

# app/services/procurement_service.py

class PurchasePlanMergeService:
    
    @staticmethod
    def group_by_item(
        pcplanids: list[str]
    ) -> dict[str, list[RequisitionDetail]]:
        """
        按物料编码分组，用于并单
        
        Returns:
            {itemcd: [detail1, detail2, ...]}
        """
        # 查询所有需求明细
        details = PurchasePlanRepository.get_details_by_pcplanids(pcplanids)
        
        # 按itemcd分组
        groups = defaultdict(list)
        for d in details:
            if d.auditqty > 0:  # 只合并有审核数量的
                groups[d.itemcd].append(d)
        
        return groups
    
    @staticmethod
    def generate_merge_preview(
        pcplanids: list[str]
    ) -> list[MergeGroup]:
        """
        生成合并预览
        """
        groups = PurchasePlanMergeService.group_by_item(pcplanids)
        
        preview = []
        for itemcd, details in groups.items():
            total_qty = sum(d.auditqty for d in details)
            
            preview.append({
                "itemcd": itemcd,
                "itemnm": details[0].itemnm,
                "total_qty": total_qty,
                "source_lines": [
                    {
                        "pcplanid": d.pcplanid,
                        "pclineno": d.lineno,
                        "qty": d.auditqty,
                        "dept": d.dept_nm
                    }
                    for d in details
                ],
                "suggested_suppliers": SupplierService.get_suppliers_by_item(itemcd)
            })
        
        return preview

# app/services/procurement_service.py

class PurchaseRegisterService:
    
    @staticmethod
    def merge_create(
        merge_groups: list[dict[str, Any]],
        memo: str | None,
        creator: str
    ) -> dict[str, Any]:
        """
        合并多个需求单生成采购订单
        """
        created_orders = []
        total_links = 0
        
        for group in merge_groups:
            itemcd = group.get("itemcd")
            suppliercd = group.get("suppliercd")
            lines = group.get("lines", [])
            
            if not lines:
                continue
            
            # 1. 校验所有需求行可用数量
            for line in lines:
                available = PurchasePlanRepository.get_available_qty(
                    line["pcplanid"], line["pclineno"]
                )
                if line["qty"] > available:
                    raise ValueError(
                        f"需求 {line['pcplanid']} 行 {line['pclineno']} "
                        f"可用数量不足：需要{line['qty']}，实际{available}"
                    )
            
            # 2. 创建订单主表
            total_qty = sum(line["qty"] for line in lines)
            order_data = {
                "suppliercd": suppliercd,
                "memo": f"{memo or '合并采购'} - {itemcd}",
            }
            order = PurchaseRegisterRepository.create(order_data, creator)
            rgstbillid = order.rgstbillid
            
            # 3. 创建订单明细（合并同类商品）
            PurchaseRegisterRepository.add_detail(
                rgstbillid=rgstbillid,
                lineno=1,
                data={
                    "itemcd": itemcd,
                    "rgsqty": total_qty,
                    "ref_pcplanid": lines[0]["pcplanid"],  # 主要来源
                    "ref_pclineno": lines[0]["pclineno"]
                }
            )
            
            # 4. 创建多条TPC20关联（关键）
            link_details = [
                {
                    "ref_pcplanid": line["pcplanid"],
                    "ref_pclineno": line["pclineno"],
                    "rgstbillid": rgstbillid,
                    "rgstlineno": 1,
                    "rgsqty": line["qty"],
                    "itemcd": itemcd
                }
                for line in lines
            ]
            RequisitionOrderLinkRepository.create_links(link_details)
            total_links += len(lines)
            
            created_orders.append(rgstbillid)
        
        db.session.commit()
        return {
            "created_orders": created_orders,
            "linked_requisitions": total_links,
            "count": len(created_orders)
        }
```

  selectedSupplier: string;
  price: number;
}

// 合并预览弹窗
const mergePreviewVisible = ref(false);
const mergeGroups = ref<MergePreviewGroup[]>([]);
const selectedRequisitions = ref<string[]>([]);

// 点击合并按钮
const handleMergePreview = async () => {
  if (selectedRequisitions.value.length < 2) {
    message.warning('请至少选择2个申请单');
    return;
  }
  
  // 调用预览API
  const response = await getMergePreview({
    pcplanids: selectedRequisitions.value
  });
  
  mergeGroups.value = response.data.groups;
  mergePreviewVisible.value = true;
};

// 确认合并创建
const handleMergeCreate = async () => {
  const mergeData = {
    merge_groups: mergeGroups.value.map(g => ({
      itemcd: g.itemcd,
      suppliercd: g.selectedSupplier,
      lines: g.sourceLines.map(l => ({
        pcplanid: l.pcplanid,
        pclineno: l.pclineno,
        qty: l.qty
      }))
    })),
    memo: '合并采购'
  };
  
  const response = await mergeCreateOrders(mergeData);
  
  if (response.code === 201) {
    message.success(
      `成功创建${response.data.count}个订单，` +
      `关联${response.data.linked_requisitions}条需求明细`
    );
    mergePreviewVisible.value = false;
    fetchList(); // 刷新列表
  }
};
```

## 五、数据模型与约束

### 5.1 TPC20关联表约束验证

两个场景都依赖TPC20的UK约束：
```sql
UNIQUE (pcplanid, pclineno, rgstbillid, rgstlineno)
```

**场景A（拆单）验证**：
```
PP000180-行1 → PO001-行1 ✓
PP000180-行1 → PO002-行1 ✓  （同一需求行+不同订单，符合UK）
```

**场景B（并单）验证**：
```
PP000180-行1 → PO001-行1 ✓
PP000181-行1 → PO001-行1 ✓  （不同需求行+同一订单，符合UK）
```

### 5.2 视图兼容性

`v_requisition_execution` 视图自动支持新功能：
```sql
-- 拆单场景：同一需求行有多个关联，SUM(linkqty)自动汇总
SELECT ... SUM(l.linkqty) ... 
FROM tpc20_requisition_order_link l
GROUP BY l.pcplanid, l.pclineno
```

### 5.3 数据库无需变更

✅ `tpc12_register` 保持不变
